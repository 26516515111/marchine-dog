> **For agentic workers:** REQUIRED SUB-SKILL: Use superpowers:subagent-driven-development (recommended) or superpowers:executing-plans to implement this plan task-by-task. Steps use checkbox (`- [ ]`) syntax for tracking.

**Goal:** Build an isolated `fire/v2.2` whose training, evaluation, export, and prediction pipelines use the same deterministic LAB-CLAHE preprocessing and no HSV input enhancement.

**Architecture:** Register an RGB-aware `CLAHEDehaze` sample operator in PaddleDetection and place it immediately after `Decode` in every V2.2 reader. PaddleDetection exports the operator and its parameters to `infer_cfg.yml`; the isolated V2.2 predictor recognizes that metadata and applies an equivalent RGB CLAHE operator exactly once.

**Tech Stack:** Python 3, OpenCV, NumPy, PaddleDetection, Paddle Inference, PyYAML

@register_op
class CLAHEDehaze(BaseOperator):
    """Deterministic LAB-L CLAHE preprocessing for decoded RGB images."""

    def __init__(self, clip_limit=3.0, tile_grid_size=(8, 8)):
        super(CLAHEDehaze, self).__init__()
        if not isinstance(clip_limit, (int, float)) or clip_limit <= 0:
            raise ValueError("clip_limit must be a positive number")
        if not isinstance(tile_grid_size, (list, tuple)) or len(
                tile_grid_size) != 2:
            raise TypeError("tile_grid_size must contain two positive integers")
        if any(
                not isinstance(value, int) or isinstance(value, bool) or
                value <= 0 for value in tile_grid_size):
            raise ValueError(
                "tile_grid_size must contain two positive integers")
        self.clip_limit = float(clip_limit)
        self.tile_grid_size = tuple(tile_grid_size)

    def apply(self, sample, context=None):
        image = sample['image']
        if not isinstance(image, np.ndarray):
            raise TypeError("{}: image is not a numpy array.".format(self))
        if image.ndim != 3 or image.shape[2] != 3:
            raise ImageError("{}: image must have shape (H, W, 3).".format(
                self))
        if image.dtype != np.uint8:
            raise TypeError("{}: image dtype must be uint8.".format(self))

        lab = cv2.cvtColor(image, cv2.COLOR_RGB2LAB)
        clahe = cv2.createCLAHE(
            clipLimit=self.clip_limit,
            tileGridSize=self.tile_grid_size)
        lab[:, :, 0] = clahe.apply(lab[:, :, 0])
        sample['image'] = cv2.cvtColor(lab, cv2.COLOR_LAB2RGB)
        return sample
```

    if spec is None or spec.loader is None:
        raise RuntimeError("cannot load V2.2 predictor")
    module = importlib.util.module_from_spec(spec)
    spec.loader.exec_module(module)

    image = _sample_rgb()

        clip_limit=3.0,
        tile_grid_size=[8, 8],

        clip_limit=3.0,

class CLAHEDehaze:
    """LAB-L CLAHE preprocessing for RGB images."""

    def __init__(
        self,
        clip_limit: float = 3.0,
        tile_grid_size: list[int] | tuple[int, int] = (8, 8),
    ):
        if not isinstance(clip_limit, (int, float)) or clip_limit <= 0:
            raise ValueError("clip_limit must be a positive number")
        if not isinstance(tile_grid_size, (list, tuple)) or len(tile_grid_size) != 2:
            raise TypeError("tile_grid_size must contain two positive integers")
        if any(
            not isinstance(value, int) or isinstance(value, bool) or value <= 0
            for value in tile_grid_size
        ):
            raise ValueError("tile_grid_size must contain two positive integers")
        self.clip_limit = float(clip_limit)
        self.tile_grid_size = tuple(tile_grid_size)

    def __call__(self, img, im_info):
        import cv2
        import numpy as np

        if not isinstance(img, np.ndarray):
            raise TypeError("image must be a numpy array")
        if img.ndim != 3 or img.shape[2] != 3:
            raise ValueError("image must have shape (H, W, 3)")
        if img.dtype != np.uint8:
            raise TypeError("image dtype must be uint8")

        lab = cv2.cvtColor(img, cv2.COLOR_RGB2LAB)
        clahe = cv2.createCLAHE(
            clipLimit=self.clip_limit,
            tileGridSize=self.tile_grid_size,
        )
        lab[:, :, 0] = clahe.apply(lab[:, :, 0])
        return cv2.cvtColor(lab, cv2.COLOR_LAB2RGB), im_info

def build_preprocess_ops(preprocess_infos: list[dict[str, Any]]) -> list[Any]:
    op_map = {
        "CLAHEDehaze": CLAHEDehaze,
        "Resize": Resize,
        "NormalizeImage": NormalizeImage,
        "Permute": Permute,
        "PadStride": PadStride,
    }
    ops = []
    clahe_count = 0
    for info in preprocess_infos:
        op_type = info["type"]
        if op_type not in op_map:
            raise ValueError("unsupported V2.2 preprocess operator: {}".format(op_type))
        if op_type == "CLAHEDehaze":
            clahe_count += 1
        params = {key: value for key, value in info.items() if key != "type"}
        ops.append(op_map[op_type](**params))
    if clahe_count != 1:
        raise ValueError(
            "V2.2 infer_cfg.yml must contain exactly one CLAHEDehaze operator"
        )
    return ops
```

from __future__ import annotations

from pathlib import Path

import yaml

PROJECT_ROOT = Path(__file__).resolve().parents[2]
CONFIG_PATH = PROJECT_ROOT / "fire" / "v2.2" / "B_fire_atmosphere_v2_2.yml"
REMOVED_OPS = {
    "AtmosphereFireEnhance",
    "BackgroundDesaturate",
    "OrangeBloom",
    "UnsharpMask",
}


class PaddleLoader(yaml.SafeLoader):
    pass


def _construct_paddle_tag(loader, _suffix, node):
    if isinstance(node, yaml.MappingNode):
        return loader.construct_mapping(node, deep=True)
    if isinstance(node, yaml.SequenceNode):
        return loader.construct_sequence(node, deep=True)
    return loader.construct_scalar(node)


PaddleLoader.add_multi_constructor("!", _construct_paddle_tag)


def _load_config() -> dict:
    return yaml.load(CONFIG_PATH.read_text(encoding="utf-8"), Loader=PaddleLoader)


def _transform_name(transform: dict) -> str:
    return next(iter(transform))


def test_all_readers_use_the_same_clahe_immediately_after_decode() -> None:
    config = _load_config()
    for reader_name in ("TrainReader", "EvalReader", "TestReader"):
        transforms = config[reader_name]["sample_transforms"]
